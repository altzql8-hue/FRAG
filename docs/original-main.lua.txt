-- ========================= HELPERS ============================
local function color(hex)
  return Color.parseColor(hex)
end

local function trim(value)
  value = tostring(value or "")
  return value:match("^%s*(.-)%s*$") or ""
end

local function timeNow()
  return os.time() * 1000
end

local function shortDeviceId(value)
  value = tostring(value or "")
  if #value <= 8 then return value end
  return "••••" .. value:sub(-8)
end

local function maskKey(value)
  value = tostring(value or "")
  if #value <= 8 then return value end
  return value:sub(1, 4) .. string.rep("•", math.min(8, #value - 8)) .. value:sub(-4)
end

local function makeDrawable(fill, stroke, radius, strokeWidth)
  local gd = GradientDrawable()
  gd.setColor(color(fill))
  gd.setCornerRadius(tonumber(radius or 18))
  if stroke then
    gd.setStroke(tonumber(strokeWidth or 1), color(stroke))
  end
  return gd
end

local function setRounded(view, fill, stroke, radius, strokeWidth)
  if view then
    view.setBackground(makeDrawable(fill, stroke, radius, strokeWidth))
  end
end

local function fadeIn(view, duration)
  if not view then return end
  local anim = AlphaAnimation(0.0, 1.0)
  anim.setDuration(duration or 450)

end

local function pulse(view)
  if not view then return end

  1.0, 0.97,

  Animation.RELATIVE_TO_SELF, 0.5,

  )
  anim.setDuration(90)
  anim.setRepeatCount(1)
  anim.setRepeatMode(Animation.REVERSE)
  view.startAnimation(anim)
end

-- ========================= SOUND ==============================
local function removePlayer(player)
  for i = #activePlayers, 1, -1 do
    if activePlayers[i] == player then
      table.remove(activePlayers, i)
      break
    end
  end
end

local function playSfx(name)
  if not ENABLE_SFX then return end

  pcall(function()
    local soundPath = activity.getLuaDir("sounds") .. "/" .. tostring(name) .. ".wav"
    local file = File(soundPath)
    if not file.exists() then return end

    local player = MediaPlayer()
    table.insert(activePlayers, player)
    player.setDataSource(soundPath)
    player.setOnCompletionListener(MediaPlayer.OnCompletionListener{
      onCompletion = function(mp)
        removePlayer(mp)
        pcall(function() mp.release() end)
      end
    })
    player.prepare()
    player.start()
  end)
end

-- ========================= ASYNC REST WRITE ====================
-- Uses a real AndLua background task for blocking HttpURLConnection writes.
local function asyncWrite(method, url, payload, callback)
  local worker = activity.newTask(
  function(reqMethod, reqUrl, reqBody)
    require "import"
    import "java.net.URL"
    import "java.io.OutputStreamWriter"

    local ok, responseCode = pcall(function()
      local conn = URL(reqUrl).openConnection()

      if reqMethod == "PATCH" then
        conn.setRequestMethod("POST")
        conn.setRequestProperty("X-HTTP-Method-Override", "PATCH")
       else
        conn.setRequestMethod(reqMethod)
      end

      conn.setConnectTimeout(10000)
      conn.setReadTimeout(10000)
      conn.setDoOutput(true)
      conn.setRequestProperty("Content-Type", "application/json; charset=UTF-8")

      local writer = OutputStreamWriter(conn.getOutputStream(), "UTF-8")
      writer.write(tostring(reqBody or ""))
      writer.flush()
      writer.close()

      local code = conn.getResponseCode()
      conn.disconnect()
      return code
    end)

    if not ok then
      return false, -1
    end

    local code = tonumber(responseCode) or -1
    return (code >= 200 and code < 300), code
  end,
  nil,
  function(ok, code)
    if callback then callback(ok == true, tonumber(code) or -1) end
  end
  )

  worker.execute{method, url, payload}
end

-- ========================= LOCKOUT =============================
function lockLoginSystem()
  if isLocked then return end

  isLocked = true
  requestInFlight = false
  setLoginButton(false, "TEMPORARILY LOCKED")
  playSfx("locked")

  local timeLeft = lockDuration
  local function countdown()
    if timeLeft > 0 then
      logMsg("LOCKED • TRY AGAIN IN " .. timeLeft .. "S", COLORS.red)
      timeLeft = timeLeft - 1
      task(1000, countdown)
     else
      isLocked = false
      setLoginButton(true)
      logMsg("SYSTEM READY", COLORS.muted)
      customToast("LOGIN UNLOCKED", COLORS.cyan)
      lockDuration = math.min(lockDuration + 5, 30)
    end
  end

  countdown()
end

-- ========================= LOGIN LOGIC =========================
function finalizeLogin(key, data, currentDeviceCount)
  requestInFlight = false
  wrongAttempts = 0
  setLoginButton(true)
  logMsg("AUTHENTICATION SUCCESSFUL", COLORS.green)
  customToast("AUTHENTICATION SUCCESSFUL", COLORS.green)
  speak("Authentication successful")

  local sp = activity.getSharedPreferences("FragHaxPrefs", 0)
  local editor = sp.edit()

  if rememberCb and rememberCb.Checked then
    editor.putString("saved_key", key)
    editor.putBoolean("remember_me", true)
   else
    editor.remove("saved_key")
    editor.putBoolean("remember_me", false)
  end
  editor.apply()

  local currentUsage = tonumber(data.usageCount) or 0
  local newUsage = currentUsage + 1
  local patchUrl = DB_URL .. "/keys/" .. key .. ".json"
  local patchData = cjson.encode({
    usageCount = newUsage,
    lastUsed = timeNow()
  })

  -- Usage logging should never block successful login.
  asyncWrite("PATCH", patchUrl, patchData, function() end)

  local expiryText = "Lifetime"
  local expiresAt = tonumber(data.expiresAt) or 0
  if expiresAt > 0 then
    expiryText = os.date("%Y-%m-%d %H:%M", math.floor(expiresAt / 1000))
  end

  showSuccessDialog(
  key,
  string.upper(tostring(data.type or "PREMIUM")),
  expiryText,
  currentDeviceCount,
  tonumber(data.maxDevices) or 1,
  newUsage
  )
end

local function validateKeyFormat(key)
  if key == "" then
    return false, "KEY REQUIRED"
  end

  if #key < 3 or #key > 120 then
    return false, "INVALID KEY FORMAT"
  end

  -- Firebase RTDB node names cannot contain these characters.
  if key:find("[%.#$%[%]/]") then
    return false, "INVALID KEY FORMAT"
  end

  return true
end

local function beginLogin()
  if isLocked or requestInFlight then return end

  local key = trim(keyInput and keyInput.Text or "")
  local valid, reason = validateKeyFormat(key)

  if not valid then
    logMsg("ERROR • " .. reason, COLORS.red)
    customToast(reason, COLORS.red)
    playSfx("error")
    speak(reason)
    return
  end

  requestInFlight = true
  setLoginButton(false, "VERIFYING…")
  logMsg("CONTACTING SERVER…", COLORS.cyan)
  playSfx("tap")

  Http.get(DB_URL .. "/keys/" .. key .. ".json", function(code, content)
    local responseCode = tonumber(code) or -1

    if responseCode ~= 200 then
      requestInFlight = false
      setLoginButton(true)
      logMsg("CONNECTION ERROR • " .. tostring(responseCode), COLORS.red)
      customToast("CONNECTION ERROR", COLORS.red)
      playSfx("error")
      speak("Connection error")
      return
    end

    if not content or content == "" or content == "null" then
      reportError("ACCESS DENIED • INVALID KEY", "INVALID LICENSE KEY")
      return
    end

    local decodeOk, data = pcall(cjson.decode, content)
    if not decodeOk or type(data) ~= "table" then
      requestInFlight = false
      setLoginButton(true)
      logMsg("SERVER RESPONSE ERROR", COLORS.red)
      customToast("INVALID SERVER RESPONSE", COLORS.red)
      playSfx("error")
      return
    end

    local expiresAt = tonumber(data.expiresAt) or 0
    if expiresAt > 0 and timeNow() > expiresAt then
      reportError("ACCESS DENIED • KEY EXPIRED", "KEY EXPIRED", "expired")
      return
    end

    local status = tostring(data.status or ""):lower()
    if status ~= "active" then
      if status == "banned" then
        reportError("ACCESS DENIED • BANNED", "LICENSE BANNED", "banned")
       elseif status == "frozen" then
        reportError("ACCESS DENIED • FROZEN", "LICENSE FROZEN", "frozen")
       elseif status == "expired" then
        reportError("ACCESS DENIED • EXPIRED", "LICENSE EXPIRED", "expired")
       else
        reportError("ACCESS DENIED • " .. string.upper(status ~= "" and status or "INACTIVE"), "ACCESS DENIED")
      end
      return
    end

    local hwids = type(data.hwids) == "table" and data.hwids or {}
    local isMatched = false
    local count = 0

    for _, value in pairs(hwids) do
      count = count + 1
      if tostring(value) == tostring(hwid) then
        isMatched = true
      end
    end

    local maxDevices = math.max(1, tonumber(data.maxDevices) or 1)

    if isMatched then
      finalizeLogin(key, data, count)
      return
    end

    if count >= maxDevices then
      reportError("DEVICE LIMIT REACHED", "DEVICE LIMIT REACHED")
      return
    end

    logMsg("REGISTERING DEVICE…", COLORS.cyan)
    local node = "hwid_" .. tostring(count + 1)
    local regUrl = DB_URL .. "/keys/" .. key .. "/hwids/" .. node .. ".json"
    local payload = cjson.encode(hwid)

    asyncWrite("PUT", regUrl, payload, function(ok)
      if ok then
        finalizeLogin(key, data, count + 1)
       else
        requestInFlight = false
        setLoginButton(true)
        logMsg("DEVICE REGISTRATION FAILED", COLORS.red)
        customToast("DEVICE REGISTRATION FAILED", COLORS.red)
        playSfx("error")
      end
    end)
  end)
end

function setupLoginLogic()
  setRounded(brandMark, COLORS.panel2, COLORS.cyanSoft, 72, 2)
  setRounded(loginPanel, COLORS.panel, COLORS.border, 24, 1)
  setRounded(keyFieldShell, COLORS.field, COLORS.border, 16, 1)
  setRounded(loginBtn, COLORS.cyan, COLORS.cyan, 16, 1)
  setRounded(statusShell, COLORS.field, COLORS.border, 14, 1)

  keyInput.setSingleLine(true)
  keyInput.setInputType(InputType.TYPE_CLASS_TEXT)
  keyInput.setImeOptions(6) -- EditorInfo.IME_ACTION_DONE

  buyKeyText.onClick = function()
    playSfx("tap")
    openUrl(CONTACT_URL)
  end

  clearKeyBtn.onClick = function()
    playSfx("tap")
    keyInput.setText("")
    keyInput.requestFocus()
  end

  loginBtn.onClick = function()
    pulse(loginBtn)
    beginLogin()
  end

  keyInput.setOnEditorActionListener(TextView.OnEditorActionListener{
    onEditorAction = function(_, actionId, _)
      if actionId == 6 then
        beginLogin()
        return true
      end
      return false
    end
  })
end

  activity.setContentView(view)
  setupLoginLogic()

  local sp = activity.getSharedPreferences("FragHaxPrefs", 0)
  if sp.getBoolean("remember_me", false) then
    keyInput.setText(sp.getString("saved_key", ""))
    rememberCb.Checked = true
  end

  fadeIn(view, 500)
  customToast("WELCOME TO FRAG HAX", COLORS.cyan)
  speak("Welcome to Frag Hax login")
end)

-- ========================= CLEANUP =============================
function onDestroy()
  pcall(function()
    if tts then
      tts.stop()
      tts.shutdown()
      tts = nil
    end
  end)

  for i = #activePlayers, 1, -1 do
    local player = activePlayers[i]
    pcall(function()
      player.stop()
      player.release()
    end)
    activePlayers[i] = nil
  end
end
